# 1.0.08 - Install Spatie Permission & RBAC System

## Objective
Install spatie/laravel-permission, configure RBAC with roles super-admin, admin, students, teacher, staff, accounts, guest, add User Role & Permissions Management UI (Permissions Management, Role Assignment with permission checkboxes, User-Role Mapping), use table prefix tbl_rbac_, and add modern responsive tables.

## Commands Run (from laravel-web-api/apps/)
`ash
composer require spatie/laravel-permission
php artisan vendor:publish --provider="Spatie\Permission\PermissionServiceProvider"
php artisan migrate
php artisan db:seed --force
php artisan optimize:clear
php artisan permission:cache-reset
`

<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Spatie\Permission\Models\Permission;

class PermissionController extends Controller
{
    public function index()
    {
         = Permission::all();
        return view('admin.permissions.index', compact('permissions'));
    }

    public function create()
    {
        return view('admin.permissions.create');
    }

    public function store(Request )
    {
        ->validate([
            'name' => 'required|string|unique:permissions,name',
        ]);

        Permission::create([
            'name' => ->name,
            'guard_name' => 'web',
        ]);

        return redirect()->route('admin.permissions.index')->with('success', 'Permission created successfully.');
    }

    public function edit(Permission )
    {
        return view('admin.permissions.edit', compact('permission'));
    }

    public function update(Request , Permission )
    {
        ->validate([
            'name' => 'required|string|unique:permissions,name,' . ->id,
        ]);

        ->update([
            'name' => ->name,
        ]);

        return redirect()->route('admin.permissions.index')->with('success', 'Permission updated successfully.');
    }

    public function destroy(Permission )
    {
        ->delete();
        return redirect()->route('admin.permissions.index')->with('success', 'Permission deleted successfully.');
    }
}

### app/Http/Controllers/Admin/RoleController.php
`php
<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;
use Spatie\Permission\Models\Role;
use Spatie\Permission\Models\Permission;

class RoleController extends Controller
{
    public function index()
    {
         = Role::with('permissions')->get();
        return view('admin.roles.index', compact('roles'));
    }

    public function create()
    {
         = Permission::all();
        return view('admin.roles.create', compact('permissions'));
    }

    public function store(Request )
    {
        ->validate([
            'name' => 'required|string|unique:roles,name',
            'permissions' => 'array',
        ]);

         = Role::create([
            'name' => ->name,
            'guard_name' => 'web',
        ]);

        if (->has('permissions')) {
            ->syncPermissions(->permissions);
        }

        return redirect()->route('admin.roles.index')->with('success', 'Role created successfully.');
    }

    public function show(Role )
    {
        ->load('permissions');
        return view('admin.roles.show', compact('role'));
    }

    public function edit(Role )
    {
         = Permission::all();
         = ->permissions->pluck('name')->toArray();
        return view('admin.roles.edit', compact('role', 'permissions', 'rolePermissions'));
    }

    public function update(Request , Role )
    {
        ->validate([
            'name' => 'required|string|unique:roles,name,' . ->id,
            'permissions' => 'array',
        ]);

        ->update([
            'name' => ->name,
        ]);

        if (->has('permissions')) {
            ->syncPermissions(->permissions);
        } else {
            ->syncPermissions([]);
        }

        return redirect()->route('admin.roles.index')->with('success', 'Role updated successfully.');
    }

    public function destroy(Role )
    {
        if (->name === 'super-admin') {
            return redirect()->route('admin.roles.index')->with('error', 'Cannot delete super-admin role.');
        }

        ->delete();
        return redirect()->route('admin.roles.index')->with('success', 'Role deleted successfully.');
    }
}

<?php

namespace Database\Seeders;

use Illuminate\Database\Seeder;
use Spatie\Permission\Models\Role;
use Spatie\Permission\Models\Permission;

class RolesAndPermissionsSeeder extends Seeder
{
    /**
     * Run the database seeds.
     */
    public function run(): void
    {
        // Reset cached roles and permissions
        app()[\Spatie\Permission\PermissionRegistrar::class]->forgetCachedPermissions();

        // Define permissions
         = [
            // User Management
            'user.view',
            'user.create',
            'user.edit',
            'user.delete',

            // Role Management
            'role.view',
            'role.create',
            'role.edit',
            'role.delete',
            'role.assign-permissions',

            // Permission Management
            'permission.view',
            'permission.create',
            'permission.edit',
            'permission.delete',

            // Dashboard
            'dashboard.access',
        ];

        foreach ( as ) {
            Permission::firstOrCreate(['name' => , 'guard_name' => 'web']);
        }

        // Create roles
         = [
            'super-admin',
            'admin',
            'students',
            'teacher',
            'staff',
            'accounts',
            'guest',
        ];

        foreach ( as ) {
            Role::firstOrCreate(['name' => , 'guard_name' => 'web']);
        }

        // Assign all permissions to super-admin
         = Role::findByName('super-admin');
        ->syncPermissions(Permission::all());

        // Assign permissions to admin
         = Role::findByName('admin');
        ->syncPermissions([
            'user.view',
            'user.create',
            'user.edit',
            'user.delete',
            'role.view',
            'role.assign-permissions',
            'permission.view',
            'permission.create',
            'permission.edit',

## Notes
- Spatie tables use prefix: tbl_rbac_permissions, tbl_rbac_roles, tbl_rbac_model_has_permissions, tbl_rbac_model_has_roles, tbl_rbac_role_has_permissions
- Primary roles: super-admin, admin, students, teacher, staff, accounts, guest
- UI: Permissions Management (create/assign new Permission), Role Assignment (permission checkboxes), User-Role Mapping
- Default seeded super-admin: admin@example.com (password: password)
